  // Do not void a business/assertion failure merely because an unrelated
  // request failed earlier or during teardown. The external request must be
  // the unrecovered final request, must have driven the Turn to `failed`, or
  // must be the exact model failure rejected by the test assertion.

      || (lastTerminal.phase === "stream_timed_out" && lastTerminal.reason === "proxy_timeout")

        && isRetryableUpstreamStatus(lastTerminal.status))
