export function prepareSource(repository, destination, masterSha, suiteSha) {
  const overlay = evaluationTestOverlay(repository, suiteSha);
  git(repository, "clone", "--no-hardlinks", "--no-checkout", repository, destination);
  git(destination, "checkout", "--detach", masterSha);
  git(destination, "restore", `--source=${suiteSha}`, "--staged", "--worktree", "--", ...overlay);
  assertTestOnlyOverlay(git(destination, "diff", "--name-only", masterSha, "--").split("\n").filter(Boolean), overlay);
  return git(destination, "write-tree");
}
