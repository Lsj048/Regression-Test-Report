      const args = objectValue(raw, "TaskCreate");
      const now = new Date().toISOString();
      const activeForm = optionalString(args, "activeForm");
      const task: StoredTask = { id: String(state.tasks.reduce((maximum, item) => Math.max(maximum, Number(item.id) || 0), 0) + 1), subject: stringValue(args, "subject"), description: stringValue(args, "description"), ...(activeForm === undefined ? {} : { activeForm }), status: "pending", metadata: isObject(args["metadata"]) ? { ...args["metadata"] } : {}, blocks: [], blockedBy: [], createdAt: now, updatedAt: now };
      state.tasks.push(task);
      return { task } as unknown as JsonValue;
    }),
