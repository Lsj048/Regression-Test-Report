const object = (
  properties: Readonly<Record<string, JsonSchema>>,
  required: readonly string[] = [],
): JsonSchema => ({
  type: "object",
  properties,
  ...(required.length === 0 ? {} : { required }),
  additionalProperties: false,
});

const string = (description: string, extra: JsonSchema = {}): JsonSchema => ({ type: "string", description, ...extra });
const boolean = (description: string): JsonSchema => ({ type: "boolean", description });
const integer = (description: string, extra: JsonSchema = {}): JsonSchema => ({ type: "integer", description, ...extra });

const schemaByName = new Map(FILESYSTEM_SEARCH_TOOL_SCHEMAS.map((schema) => [schema.name, schema]));

function text(value: JsonValue): readonly ContentBlock[] {
  return [{ type: "text", text: typeof value === "string" ? value : JSON.stringify(value) }];
}

export function filesystemSearchTool(
  name: FilesystemSearchToolName,
  options: {
    readonly execute: (args: JsonValue, context: ToolTurnContext) => Promise<JsonValue>;
    readonly permission: (args: JsonValue) => ToolPermissionTarget;
    readonly parallel?: boolean;
    readonly render?: (args: JsonValue, value: JsonValue) => readonly ContentBlock[];
  },
): ToolDefinition {
  const schema = schemaByName.get(name);
  if (schema === undefined) throw new Error(`Unknown filesystem search Tool schema: ${name}`);
  return Object.freeze({
    ...schema,
    constraintEnforcers: [RESOURCE_SCOPE_ENFORCER],
    output: { schema: {}, render: options.render ?? ((_args: JsonValue, value: JsonValue) => text(value)) },
    execute: options.execute,
    permission: options.permission,
    ...(options.parallel === undefined ? {} : { isConcurrencySafe: () => options.parallel === true }),
  });
}

export function objectValue(value: JsonValue, name: string): Readonly<Record<string, JsonValue>> {
  if (!value || typeof value !== "object" || Array.isArray(value)) throw new Error(`${name} must be an object`);
  return value as Readonly<Record<string, JsonValue>>;
}

export function stringValue(value: Readonly<Record<string, JsonValue>>, key: string): string {
  const item = value[key];
  if (typeof item !== "string") throw new Error(`${key} must be a string`);
  return item;
}

export function optionalString(value: Readonly<Record<string, JsonValue>>, key: string): string | undefined {
  const item = value[key];
  return typeof item === "string" ? item : undefined;
}

export function optionalNumber(value: Readonly<Record<string, JsonValue>>, key: string): number | undefined {
  const item = value[key];
  return typeof item === "number" && Number.isFinite(item) ? item : undefined;
}
