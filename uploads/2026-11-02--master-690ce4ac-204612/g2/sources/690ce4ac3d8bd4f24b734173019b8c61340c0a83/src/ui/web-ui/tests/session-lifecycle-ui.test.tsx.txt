import { describe, expect, it } from "vitest";
import { renderToStaticMarkup } from "react-dom/server";
import { ModelSelect, SessionIdentity, sessionCanStop, sessionSummaryAfterEvent } from "../client/App.js";
import { pendingUserQuestions } from "../shared/user-questions.js";
import type { WebEvent, WebSessionSummary } from "../shared/protocol.js";

const createdAt = "2026-09-01T10:00:00.000Z";

describe("session lifecycle UI", () => {
  it("shows an actionable model selector and explains when a task prevents switching", () => {
    const render = (disabled: boolean) => renderToStaticMarkup(<ModelSelect value="glm-5v-turbo" models={["glm-5v-turbo", "glm-5.3"]} disabled={disabled} disabledReason={disabled ? "请先停止当前任务" : undefined} onChange={() => undefined} />);
    expect(render(false)).toContain('data-testid="model-select"');
    expect(render(false)).not.toContain('disabled=""');
    expect(render(true)).toContain('disabled=""');
    expect(render(true)).toContain("请先停止当前任务");
  });

  it.each(["session-1", "child-session"])("updates the model only from the current session's durable event (%s)", (sourceSessionId) => {
    const session = { id: "session-1", model: "glm-5v-turbo", updatedAt: createdAt, lastCursor: 10 } as WebSessionSummary;
    const event: WebEvent = { cursor: 11, eventId: "model-change", sessionId: session.id, sourceSessionId, type: "session.updated", payload: { to: { model: "glm-5.3" } }, payloadHash: "hash", traceId: null, createdAt };
    expect(sessionSummaryAfterEvent(session, event).model).toBe(sourceSessionId === session.id ? "glm-5.3" : "glm-5v-turbo");
  });

  const questionEvent = (kind?: "user_input" | "approval"): WebEvent => ({
    cursor: 1, eventId: "question-1", sessionId: "root", sourceSessionId: "root", turnId: "turn-1",
    type: "host.question", payload: { requestId: "request-1", ...(kind ? { kind } : {}),
      questions: [{ question: "请选择", header: "问题", options: [], multiSelect: false }] },
    payloadHash: "hash", traceId: null, createdAt,
  });

  it.each(["user_input", undefined] as const)("does not offer stop for a restored ordinary question (kind=%s)", (kind) => {
    const requested = questionEvent(kind);
    const questions = pendingUserQuestions([requested, { ...requested, cursor: 2, eventId: "wait-2",
      type: "turn.waiting", payload: { expectations: [{ kind: "user_input" }] } }]);
    expect(questions).toHaveLength(1);
    expect(questions[0]?.kind).toBe("user_input");
    expect(sessionCanStop("waiting", false, questions)).toBe(false);
  });

  it("keeps stop available for a restored permission approval without marking it as running", () => {
    const questions = pendingUserQuestions([questionEvent("approval")]);
    expect(questions).toHaveLength(1);
    expect(sessionCanStop("waiting", false, questions)).toBe(true);
    for (const status of ["idle", "completed", "blocked", "failed", "interrupted"] as const) {
      expect(sessionCanStop(status, false, questions)).toBe(false);
    }
  });

  it.each(["host.question.answered", "turn.interrupted"])("removes the stop control after %s closes the approval", (type) => {
    const requested = questionEvent("approval");
    const questions = pendingUserQuestions([requested, { ...requested, cursor: 2, eventId: "closed-2",
      type, payload: { requestId: "request-1" } }]);
    expect(questions).toEqual([]);
    expect(sessionCanStop("waiting", false, questions)).toBe(false);
  });

  it("retains stop for running tasks and child work, but not an empty wait", () => {
    expect(sessionCanStop("running", true, [])).toBe(true);
    expect(sessionCanStop("waiting", true, pendingUserQuestions([questionEvent("user_input")]))).toBe(true);
    expect(sessionCanStop("waiting", false, [])).toBe(false);
  });
  it("shows the full session ID in a copy control", () => {
    const sessionId = "session_a5e33c90-7fa4-42df-b48c-9b02071dbf86";
    const html = renderToStaticMarkup(<SessionIdentity sessionId={sessionId} />);
    expect(html).toContain(sessionId);
    expect(html).toContain(`复制会话 ID ${sessionId}`);
    expect(html).toContain("session-identity");
  });
  it.each(["completed", "waiting", "failed"] as const)("updates a running session when the host reports %s", (status) => {
    const session: WebSessionSummary = {
      id: "session-1",
      title: "后台会话",
      workspace: "/tmp/workspace",
      model: "glm-5.2",
      mode: "standard",
      createdAt,
      updatedAt: createdAt,
      status: "running",
      sessionStatus: { kind: "active" },
      turnCount: 1,
      lastCursor: 10,
      permissionMode: "danger-full-access",
      pinned: false,
      archived: false,
      blank: false,
    };
    const completedAt = "2026-09-01T10:01:00.000Z";
    const event: WebEvent = {
      cursor: 11,
      eventId: "event-11",
      sessionId: session.id,
      sourceSessionId: session.id,
      turnId: "run-1",
      type: "ui.session.status.changed",
      payload: { status },
      payloadHash: "hash",
      traceId: null,
      createdAt: completedAt,
      sessionUpdatedAt: completedAt,
    };

    expect(sessionSummaryAfterEvent(session, event)).toMatchObject({
      status,
      updatedAt: completedAt,
      lastCursor: 11,
    });
  });

  it("keeps the session running at a per-Turn completion boundary", () => {
    const session = {
      id: "session-with-queued-input",
      status: "running",
      updatedAt: createdAt,
      lastCursor: 10,
    } as WebSessionSummary;
    const event = {
      cursor: 11,
      sessionId: session.id,
      sourceSessionId: session.id,
      type: "turn.completed",
      payload: {},
      sessionUpdatedAt: createdAt,
    } as WebEvent;

    expect(sessionSummaryAfterEvent(session, event).status).toBe("running");
  });

  it.each([
    ["root-session", "user_input", "waiting"],
    ["child-session", "user_input", "running"],
    ["root-session", "subagent", "running"],
  ])("projects only root user-input waits (%s, %s)", (sourceSessionId, kind, status) => {
    const session = { id: "root-session", status: "running", updatedAt: createdAt, lastCursor: 10 } as WebSessionSummary;
    const event: WebEvent = { cursor: 11, eventId: "wait-11", sessionId: session.id, sourceSessionId, type: "turn.waiting", payload: { expectations: [{ kind }] }, payloadHash: "hash", traceId: null, createdAt };
    expect(sessionSummaryAfterEvent(session, event).status).toBe(status);
  });

  it("stops a stale running session when crash recovery interrupts its run", () => {
    const session = {
      id: "crashed-session",
      status: "running",
      updatedAt: createdAt,
      lastCursor: 10,
    } as WebSessionSummary;
    const event: WebEvent = {
      cursor: 11,
      eventId: "event-crash",
      sessionId: session.id,
      sourceSessionId: session.id,
      turnId: "run-crash",
      type: "turn.interrupted",
      payload: { cause: "process_crash" },
      payloadHash: "hash",
      traceId: null,
      createdAt,
      sessionUpdatedAt: createdAt,
    };

    expect(sessionSummaryAfterEvent(session, event)).toMatchObject({ status: "interrupted", sessionStatus: { kind: "idle" } });
  });

  it("does not let a descendant completion event finish the root session", () => {
    const session = {
      id: "root-session",
      status: "running",
      updatedAt: createdAt,
      lastCursor: 10,
    } as WebSessionSummary;
    const event = {
      cursor: 11,
      sessionId: session.id,
      sourceSessionId: "child-session",
      type: "turn.completed",
      payload: {},
      sessionUpdatedAt: createdAt,
    } as WebEvent;

    expect(sessionSummaryAfterEvent(session, event).status).toBe("running");
  });
});
