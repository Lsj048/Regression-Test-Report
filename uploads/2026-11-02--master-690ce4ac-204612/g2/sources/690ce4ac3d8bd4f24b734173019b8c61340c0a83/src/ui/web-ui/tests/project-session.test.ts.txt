import { describe, expect, it } from "vitest";
import { groupTimelineItems } from "../client/timeline-groups.js";
import { projectSession, projectSubagentFlow, rootSessionStatusFromEvent, toolResultText } from "../shared/project-session.js";
import type { WebEvent, WebJson } from "../shared/protocol.js";

function event(cursor: number, type: string, payload: WebJson): WebEvent {
  return {
    cursor,
    eventId: `event-${cursor}`,
    sessionId: "session-1",
    sourceSessionId: "session-1",
    turnId: "run-1",
    type,
    payload,
    payloadHash: `hash-${cursor}`,
    traceId: null,
    createdAt: `2026-08-25T00:00:0${cursor}.000Z`,
  };
}

describe("projectSession", () => {
  for (const child of [false, true]) for (const rejected of [false, true]) {
    it(`keeps same-turn call occurrences distinct (child=${child}, rejected=${rejected})`, () => {
      const source = (cursor: number, type: string, payload: WebJson): WebEvent => ({
        ...event(cursor, type, payload), ...(child ? { sourceSessionId: "child", turnId: "child-turn" } : {}),
      });
      const events = [
        event(0, "subagent.created", { subagentId: "child", initialTurnId: "child-turn", parentSessionId: "session-1", mode: "spawn", depth: 1 }),
        source(1, "model.completed", { toolCalls: [{ id: "same", name: "Read", arguments: '{"path":"first"}' }] }),
        source(2, "tool.call", { actionPlanId: "plan1", actionId: "a", callId: "same", name: "Read", arguments: { path: "first" } }),
        source(3, "tool.result", { actionPlanId: "plan1", actionId: "a", callId: "same", status: "succeeded", content: "first result" }),
        source(4, "model.completed", { toolCalls: [{ id: "same", name: "Write", arguments: '{"path":"second","content":"new"}' }] }),
        ...(!rejected ? [source(5, "tool.call", { actionPlanId: "plan2", actionId: "b", callId: "same", name: "Write", arguments: { path: "second", content: "new" } })] : []),
        // A replayed old result while the new call is pending must not settle it.
        source(6, "tool.result", { actionPlanId: "plan1", actionId: "a", callId: "same", status: "succeeded", content: "first result" }),
        source(7, "tool.result", { actionPlanId: "plan2", ...(rejected ? { syntheticKind: "action_rejection" } : { actionId: "b" }),
          callId: "same", name: "Write", status: rejected ? "denied" : "succeeded", content: "second result" }),
      ];
      const project = (input: readonly WebEvent[]) => {
        const session = projectSession(input);
        return child ? projectSubagentFlow(input, session.subagents[0]!).flatMap(step => step.tools) : session.items.filter(item => item.kind === "tool");
      };
      const expected = [
        expect.objectContaining({ name: "Read", arguments: { path: "first" }, status: "succeeded", result: "first result" }),
        expect.objectContaining({ name: "Write", arguments: { path: "second", content: "new" }, status: rejected ? "denied" : "succeeded", result: "second result" }),
      ];
      const cold = project(events);
      expect(cold).toEqual(expected);
      expect(new Set(cold.map(tool => tool.id)).size).toBe(2);
      const delivered: WebEvent[] = [];
      for (const next of events) {
        delivered.push(next, next);
        const live = project(delivered);
        expect(live).toEqual(project(events.filter(item => item.cursor <= next.cursor)));
        if (next.cursor === 6 && !rejected) expect(live[1]).toMatchObject({ name: "Write", status: "running" });
      }
      expect(project(delivered)).toEqual(cold);
    });
  }

  it("matches legacy synthetic results to the current model step without action identity", () => {
    const events = [
      event(1, "model.completed", { toolCalls: [{ id: "same", name: "Read", arguments: '{"path":"one"}' }] }),
      event(2, "tool.result", { callId: "same", name: "Read", syntheticKind: "action_rejection", status: "denied", content: "first rejection" }),
      event(3, "model.completed", { toolCalls: [{ id: "same", name: "Write", arguments: '{"path":"two"}' }] }),
      event(4, "tool.result", { callId: "same", name: "Write", syntheticKind: "action_rejection", status: "denied", content: "second rejection" }),
    ];
    const tools = projectSession(events).items.filter(item => item.kind === "tool");
    expect(tools).toHaveLength(2);
    expect(tools[0]).toMatchObject({ arguments: { path: "one" }, result: "first rejection" });
    expect(tools[1]).toMatchObject({ arguments: { path: "two" }, result: "second rejection" });
  });

    expect(tools[0]).toMatchObject({ name: "Lookup", status: "denied", arguments: { query: "report" }, result: "Not available" });

    }

    const projected = projectSession([

      event(2, "input.promoted", { content: "检查项目" }),
      event(3, "context.plan.committed", {}),
      event(4, "model.completed", { content: "", reasoningContent: "先读取。" }),
      event(5, "message.appended", { messageId: "message:model:run-1:outcome-1", sourceType: "model.completed" }),

      event(7, "tool.result", { callId: "call-1", status: "succeeded", content: [{ type: "text", text: "hello" }] }),

    expect(projected.items.filter((item) => item.kind === "assistant").map((item) => item.id)).toEqual(["message:model:run-1:outcome-1", "message:model:run-1:outcome-2"]);

  it("keeps background subagents running until their settlement event arrives", () => {
    const running = projectSession([
      event(1, "turn.created", {}),
      event(2, "tool.call", { callId: "call-child", name: "subagent", arguments: { description: "研究扩展机制" } }),

    expect(running.items).toContainEqual(expect.objectContaining({

      subagentId: "session-child",

    expect(running.subagents).toEqual([expect.objectContaining({

      label: "研究扩展机制",
      status: "running",

      event(2, "subagent.created", {
        subagentId: "session-child",
        initialTurnId: "run-child",
        parentSessionId: "session-1",

        label: "研究扩展机制",

        depth: 1,

      event(2, "tool.call", { callId: "call-child", name: "subagent", arguments: { description: "研究扩展机制" } }),

        mode: "spawn",

  it("keeps active Turn activity before queued input and groups the promoted input with its own answer", () => {

      ...event(cursor, type, payload),

    });
    const sessionEvent = (cursor: number, type: string, payload: WebJson): WebEvent => {
      const { turnId: _turnId, ...sessionScoped } = event(cursor, type, payload);
      return sessionScoped;
    };
    const projection = projectSession([
      inTurn(1, "turn-1", "turn.created", {}),
      inTurn(2, "turn-1", "input.promoted", { inputId: "input-first", content: "第一条" }),
      inTurn(3, "turn-1", "model.request.started", {}),
      inTurn(4, "turn-1", "model.completed", { content: "", reasoningContent: "执行命令" }),
      inTurn(5, "turn-1", "tool.call", { callId: "call-1", name: "Bash", arguments: { command: "node task.mjs" } }),
      sessionEvent(6, "input.queued", { inputId: "input-next", content: "追加要求" }),
      inTurn(7, "turn-1", "tool.result", { callId: "call-1", status: "succeeded", content: "done" }),
      inTurn(8, "turn-1", "model.request.started", {}),
      inTurn(9, "turn-1", "model.completed", { content: "第一轮完成", reasoningContent: "" }),
      inTurn(10, "turn-1", "turn.completed", {}),
      inTurn(11, "turn-2", "turn.created", {}),
      inTurn(12, "turn-2", "input.promoted", { inputId: "input-next", content: "追加要求" }),
      inTurn(13, "turn-2", "model.request.started", {}),
      inTurn(14, "turn-2", "model.completed", { content: "第二轮完成", reasoningContent: "" }),
      inTurn(15, "turn-2", "turn.completed", {}),
    ]);

    const groups = groupTimelineItems(projection.items);
    expect(groups).toHaveLength(2);
    expect(groups[0]).toMatchObject({
      user: { content: "第一条" },
      final: { content: "第一轮完成" },
    });
    expect(groups[0]?.activity).toEqual(expect.arrayContaining([expect.objectContaining({ kind: "tool", name: "Bash" })]));
    expect(groups[1]).toMatchObject({
      user: { content: "追加要求" },
      final: { content: "第二轮完成" },
    });
    expect(groups[1]?.user).not.toHaveProperty("delivery");
  });

  it("does not use a historical cancelled input as the insertion point for a later Turn", () => {
    const projection = projectSession([
      event(1, "input.promoted", { inputId: "input-first", content: "第一条" }),
      event(2, "model.completed", { content: "第一轮完成", reasoningContent: "" }),
      event(3, "turn.completed", {}),
      event(4, "input.queued", { inputId: "input-cancelled", content: "已取消" }),
      event(5, "input.queue.cancelled", { inputId: "input-cancelled" }),
      event(6, "turn.interrupted", {}),
      event(7, "turn.created", {}),
      event(8, "input.promoted", { inputId: "input-third", content: "第三条" }),
      event(9, "model.completed", { content: "第三轮完成", reasoningContent: "" }),
      event(10, "turn.completed", {}),
    ]);

    expect(projection.items.map((item) => item.kind === "user" ? item.content : item.kind === "assistant" ? item.content : item.kind))
      .toEqual(["第一条", "第一轮完成", "notice", "已取消", "第三条", "第三轮完成"]);
    const groups = groupTimelineItems(projection.items);
    expect(groups).toHaveLength(4);
    expect(groups[1]?.activity).toContainEqual(expect.objectContaining({ kind: "notice", content: "任务已停止" }));
    expect(groups[2]?.user).toMatchObject({ content: "已取消", delivery: "cancelled" });
    expect(groups[3]).toMatchObject({
      user: { content: "第三条" },
      final: { content: "第三轮完成" },
    });
  });

  it("treats a committed complete transition as terminal", () => {
    const projected = projectSession(
      [
        event(1, "turn.created", {}),
        event(2, "model.request.started", {}),
        event(3, "model.completed", { content: "完成", reasoningContent: "" }),
        event(4, "state.transition.requested", {
          kind: "complete_turn",
          outcome: { reason: "assistant_stop" },
        }),
      ],
      "running",
    );

    expect(projected.status).toBe("completed");
    expect(projected.activeLabel).toBeUndefined();
  });

  it("uses the authoritative session status to clear stale activity", () => {
    const projected = projectSession(
      [
        event(1, "turn.created", {}),
        event(2, "model.request.started", {}),
        event(3, "model.completed", { content: "完成", reasoningContent: "" }),
      ],
      "completed",
    );

    expect(projected.status).toBe("completed");
    expect(projected.activeLabel).toBeUndefined();
  });

  it("applies a durable session failure emitted before a run can be created", () => {
    const projected = projectSession([
      event(1, "ui.session.status.changed", { status: "failed", message: "启动失败" }),
    ], "running");
    expect(projected.status).toBe("failed");
    expect(projected.activeLabel).toBeUndefined();
    expect(projected.items[0]).toMatchObject({ kind: "notice", tone: "error", content: "启动失败" });
  });

  it("shows a Host fault after Turn completion as a session error", () => {
    const projected = projectSession([
      event(1, "turn.created", {}),
      event(2, "model.completed", { content: "完成", reasoningContent: "" }),
      event(3, "turn.completed", {}),
      event(4, "ui.session.status.changed", { status: "failed", code: "HOST_TURN_DRIVER_FAILED", message: "final status notification unavailable" }),
    ]);
    expect(projected.status).toBe("failed");
    expect(projected.activeLabel).toBeUndefined();
    expect(projected.items).toContainEqual(expect.objectContaining({ kind: "notice", tone: "error", content: "final status notification unavailable" }));
    expect(projected.items).toContainEqual(expect.objectContaining({ kind: "assistant", content: "完成" }));
  });

  it("shows structured Host questions and records the answer without creating a new task", () => {
    const projected = projectSession([
      event(1, "turn.created", {}),

  });

  it("projects manual compaction as one persistent command card without a user message", () => {

  it("keeps descendant model traffic and internal completion notifications out of the root chat", () => {
    const child = (cursor: number, type: string, payload: WebJson): WebEvent => ({
      ...event(cursor, type, payload),
      sourceSessionId: "session-child",
      turnId: "run-child",
    });
    const projected = projectSession([
      event(1, "turn.created", {}),
      event(2, "input.promoted", { content: "启动子任务" }),
      event(3, "subagent.created", {
        subagentId: "session-child",
        initialTurnId: "run-child",
        parentSessionId: "session-1",
        label: "内部子任务",
        mode: "fork",
        depth: 1,
      }),
      child(4, "turn.created", {}),
      child(5, "input.promoted", { content: "内部子任务" }),
      child(6, "model.request.started", {}),
      child(7, "model.completed", { content: "内部结果", reasoningContent: "内部推理" }),
      child(8, "host.question", { questions: [{ question: "子任务是否允许写入？", options: [{ label: "允许" }, { label: "拒绝" }] }] }),
      child(9, "host.question.answered", { content: "允许" }),
      event(10, "subagent.settled", { subagentId: "session-child", turnId: "run-child", state: "completed", output: "内部结果" }),
      event(11, "input.promoted", {
        inputId: "input_subagent_notice:session-child:run-child",
        content: "<task-notification><result>内部结果</result></task-notification>",
        origin: {
          kind: "subagent_notification",
          subagentId: "session-child",
          childTurnId: "run-child",
          label: "内部子任务",
          status: "completed",
        },
      }),
      event(12, "state.completed", {}),
    ]);
    expect(projected.items.some((item) => item.kind === "assistant" && item.content === "内部结果")).toBe(false);
    expect(projected.items.some((item) => item.kind === "user" && item.content === "内部子任务")).toBe(false);
    expect(projected.items).toContainEqual(expect.objectContaining({ kind: "subagent", subagentId: "session-child" }));
    expect(projected.items).toContainEqual(expect.objectContaining({ kind: "notice", content: "子任务是否允许写入？\n可选：允许 / 拒绝" }));
    expect(projected.items).toContainEqual(expect.objectContaining({ kind: "user", content: "允许" }));
    expect(projected.items.some((item) => item.kind === "notice" && item.content.includes("task-notification"))).toBe(false);
    expect(projected.subagents).toEqual([expect.objectContaining({
      label: "内部子任务",
      mode: "fork",
      status: "completed",
      output: "内部结果",
      modelCalls: 1,
    })]);
  });
